cmake_minimum_required(VERSION 3.22)
project(l1nesip C CXX)
set(V ${CMAKE_SOURCE_DIR}/../../../../../vendor/android/${ANDROID_ABI})
set(P ${V}/pjsip/lib)
set(Z ${CMAKE_SOURCE_DIR}/../../../../../native/zrtp)
add_library(l1nesip SHARED l1ne_sip.c ${Z}/zrtp.c ${Z}/l1srtp.c ${Z}/l1ne_zrtp_tp.c)
target_include_directories(l1nesip PRIVATE ${V}/pjsip/include ${V}/openssl/include ${Z})
target_compile_definitions(l1nesip PRIVATE PJ_AUTOCONF=1 PJ_IS_BIG_ENDIAN=0 PJ_IS_LITTLE_ENDIAN=1)
if(ANDROID_ABI STREQUAL "armeabi-v7a")
  set(SFX armv7-unknown-linux-android)
else()
  set(SFX aarch64-unknown-linux-android)
endif()
set(LIBS pjsua pjsip-ua pjsip-simple pjsip pjmedia-codec pjmedia pjmedia-audiodev pjnath pjlib-util
         srtp resample gsmcodec speex ilbccodec g7221codec pjsdp pj)
set(ARCHIVES "")
foreach(l ${LIBS})
  list(APPEND ARCHIVES ${P}/lib${l}-${SFX}.a)
endforeach()
target_link_libraries(l1nesip -Wl,--start-group ${ARCHIVES} -Wl,--end-group ${V}/openssl/lib/libssl.a ${V}/openssl/lib/libcrypto.a -Wl,--start-group -lc++_static -lc++abi -Wl,--end-group OpenSLES mediandk log android m atomic)
